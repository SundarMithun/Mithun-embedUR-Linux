1. prime number

#include <stdio.h>
#include <pthread.h>
#include <unistd.h>

int N;

// Function to check if a number is prime
int isPrime(int num) {
    if (num < 2)
        return 0;

    for (int i = 2; i * i <= num; i++) {
        if (num % i == 0)
            return 0;
    }

    return 1;
}

// Thread A
void *sumOfPrimes(void *arg) {
    int count = 0, num = 2;
    long long sum = 0;

    while (count < N) {
        if (isPrime(num)) {
            sum += num;
            count++;
        }
        num++;
    }

    printf("Sum of first %d prime numbers = %lld\n", N, sum);
    return NULL;
}

// Thread B
void *thread1(void *arg) {
    for (int i = 0; i < 50; i++) { // 100/2 = 50 times
        printf("Thread 1 running\n");
        sleep(2);
    }
    return NULL;
}

// Thread C
void *thread2(void *arg) {
    for (int i = 0; i < 34; i++) { // approximately 100/3 times
        printf("Thread 2 running\n");
        sleep(3);
    }
    return NULL;
}

int main() {
    pthread_t A, B, C;

    printf("Enter N: ");
    scanf("%d", &N);

    pthread_create(&A, NULL, sumOfPrimes, NULL);
    pthread_create(&B, NULL, thread1, NULL);
    pthread_create(&C, NULL, thread2, NULL);

    pthread_join(A, NULL);
    pthread_join(B, NULL);
    pthread_join(C, NULL);

    return 0;
}

#include <stdio.h>
#include <pthread.h>
#include <unistd.h>
#include <signal.h>
#include <time.h>

int N;

// Signal Handler
void handleSignal(int sig) {
    printf("\nSIGINT received! Program termination prevented.\n");
}

// Prime Check Function
int isPrime(int num) {
    if (num < 2)
        return 0;

    for (int i = 2; i * i <= num; i++) {
        if (num % i == 0)
            return 0;
    }

    return 1;
}

// Thread A Function
void *sumOfPrimes(void *arg) {
    time_t start, end;
    time(&start);

    int count = 0, num = 2;
    long long sum = 0;

    printf("Thread A Started\n");

    while (count < N) {
        if (isPrime(num)) {
            sum += num;
            count++;
        }
        num++;
    }

    printf("Sum of first %d prime numbers = %lld\n", N, sum);

    time(&end);
    printf("Thread A Execution Time = %.0f seconds\n",
           difftime(end, start));

    pthread_exit(NULL);
}

// Thread B Function
void *thread1(void *arg) {
    time_t start, end;
    time(&start);

    printf("Thread B Started\n");

    for (int i = 0; i < 50; i++) {
        printf("Thread 1 running\n");
        sleep(2);
    }

    time(&end);
    printf("Thread B Execution Time = %.0f seconds\n",
           difftime(end, start));

    pthread_exit(NULL);
}

// Thread C Function
void *thread2(void *arg) {
    time_t start, end;
    time(&start);

    printf("Thread C Started\n");

    for (int i = 0; i < 34; i++) {
        printf("Thread 2 running\n");
        sleep(3);
    }

    time(&end);
    printf("Thread C Execution Time = %.0f seconds\n",
           difftime(end, start));

    pthread_exit(NULL);
}

int main() {
    pthread_t A, B, C;

    // Register signal handler
    signal(SIGINT, handleSignal);

    printf("Enter N: ");
    scanf("%d", &N);

    printf("\nCreating Threads...\n");

    pthread_create(&A, NULL, sumOfPrimes, NULL);
    pthread_create(&B, NULL, thread1, NULL);
    pthread_create(&C, NULL, thread2, NULL);

    pthread_join(A, NULL);
    pthread_join(B, NULL);
    pthread_join(C, NULL);

    printf("\nAll Threads Completed.\n");

    return 0;
}

3. short note

1. Child Process – fork()

fork() creates a new process called a child process.
The original process is called the parent process.
Both processes run independently after creation.
pid_t pid = fork();

Uses: Multitasking, process creation, command execution.


2. Handling Common Signals
Signals are software interrupts sent to a process.

Common signals:

SIGINT → Ctrl+C
SIGTERM → Terminate process
SIGKILL → Force kill
SIGSEGV → Segmentation fault
SIGCHLD → Child process terminated

Uses: Graceful shutdown, error handling, process control.


3.Exploring Different Kernel Crashes

A kernel crash (kernel panic) occurs when the OS encounters a critical error.

Common causes:

Null pointer access
Memory corruption
Faulty device drivers
Stack overflow
Deadlocks

Prevention: Proper memory management and synchronization.

4.Time Complexity

Time complexity measures how execution time grows with input size.

Complexity	Example
O(1)		Array Access
O(log n)	Binary Search
O(n)		Linear Search
O(n²)		Bubble Sort

5. Locking Mechanisms – Mutex & Spinlock
Mutex
A thread sleeps while waiting for the lock.
Suitable for longer waiting periods.
Spinlock
A thread continuously checks for the lock.
Suitable for very short waiting periods.
Mutex	Spinlock
Sleeps	Busy Waits
Less CPU Usage	More CPU Usage
Long Waits	Short Waits

Purpose: Prevent race conditions in multithreaded programs.

Conclusion:

fork() creates child processes.
Signals control and communicate with processes.
Kernel crashes occur due to critical system errors.
Time complexity measures algorithm efficiency.
Mutexes and spinlocks synchronize threads and protect shared resources
