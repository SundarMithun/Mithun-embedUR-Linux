#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define NUM_DAYS 7
#define TASK_LENGTH 100

typedef struct {
    char day_name[15];
    char **tasks;
    int num_tasks;
} Week;

int main() {
    Week days[NUM_DAYS] = {
        {"Monday", NULL, 0},
        {"Tuesday", NULL, 0},
        {"Wednesday", NULL, 0},
        {"Thursday", NULL, 0},
        {"Friday", NULL, 0},
        {"Saturday", NULL, 0},
        {"Sunday", NULL, 0}
    };

    while (1) {
        int ch, num_tasks;

        printf("\nDays:\n");
        for (int i = 0; i < NUM_DAYS; i++) {
            printf("%d. %s\n", i + 1, days[i].day_name);
        }
        printf("8. Exit\n");

        printf("Enter the day number or exit: ");
        scanf("%d", &ch);

        if (ch == 8) {
            printf("Exiting...\n");
            break;
        }

        if (ch < 1 || ch > 7) {
            printf("Invalid Day!\n");
            continue;
        }

        printf("Number of tasks: ");
        scanf("%d", &num_tasks);

        getchar();

        days[ch - 1].tasks = (char **)malloc(num_tasks * sizeof(char *));
        days[ch - 1].num_tasks = num_tasks;

        for (int i = 0; i < num_tasks; i++) {
            days[ch - 1].tasks[i] =
                (char *)malloc(TASK_LENGTH * sizeof(char));

            printf("Enter task %d: ", i + 1);
            fgets(days[ch - 1].tasks[i], TASK_LENGTH, stdin);

            days[ch - 1].tasks[i][strcspn(
                days[ch - 1].tasks[i], "\n")] = '\0';
        }
    }
    for (int i = 0; i < NUM_DAYS; i++) {
        printf("\n%s\n", days[i].day_name);

        if (days[i].num_tasks == 0) {
            printf("No tasks\n");
        } else {
            for (int j = 0; j < days[i].num_tasks; j++) {
                printf("Task %d: %s\n", j + 1, days[i].tasks[j]);
            }
        }
    }
    for (int i = 0; i < NUM_DAYS; i++) {
        for (int j = 0; j < days[i].num_tasks; j++) {
            free(days[i].tasks[j]);
        }
        free(days[i].tasks);
    }

    return 0;
}

#include<stdio.h>
#include<stdlib.h>

int main(){
    int arr[3][4]={{3,5,7,9},{4,11,21,67},{8,65,69,99}};
    int rind=0,cind=3,key,flag=0;
    printf("Enter the key:");
    scanf("%d",&key);
    while(rind<3&&cind<4){
        if(arr[rind][cind]>key){
            cind-=1;
        }
        if(arr[rind][cind]<key){
            rind+=1;
        }
        if(arr[rind][cind]==key){
            printf("Key found at index %d %d",rind,cind);
            flag=1;
            break;
        }
    }
    if(flag==0){
        printf("Key not found");
    }
    return 0;
}
