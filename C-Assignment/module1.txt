#include <stdio.h>
#include <string.h>

#define TOTAL_DAYS 7
#define MAX_TASKS_PER_DAY 3
#define MAX_TASK_LENGTH 100

// Structure to store details of a day
struct CalendarDay {
    char dayName[20];
    char taskList[MAX_TASKS_PER_DAY][MAX_TASK_LENGTH];
    int numberOfTasks;
};

int main() {
    struct CalendarDay weekCalendar[TOTAL_DAYS] = {
        {"Monday", {}, 0},
        {"Tuesday", {}, 0},
        {"Wednesday", {}, 0},
        {"Thursday", {}, 0},
        {"Friday", {}, 0},
        {"Saturday", {}, 0},
        {"Sunday", {}, 0}
    };

    int selectedDay;
    int tasksToAdd;

    printf("=== Weekly Calendar ===\n");

    while (1) {
        printf("\nChoose a day to add tasks:\n");

        for (int dayIndex = 0; dayIndex < TOTAL_DAYS; dayIndex++) {
            printf("%d. %s\n", dayIndex + 1,
                   weekCalendar[dayIndex].dayName);
        }

        printf("0. Finish Input\n");
        printf("Enter your choice: ");
        scanf("%d", &selectedDay);
        getchar(); // Remove newline character

        if (selectedDay == 0) {
            break;
        }

        if (selectedDay < 1 || selectedDay > TOTAL_DAYS) {
            printf("Invalid day choice!\n");
            continue;
        }

        selectedDay--; // Convert to array index

        printf("Enter number of tasks (maximum 3): ");
        scanf("%d", &tasksToAdd);
        getchar();

        if (tasksToAdd < 1 || tasksToAdd > MAX_TASKS_PER_DAY) {
            printf("Invalid number of tasks!\n");
            continue;
        }

        weekCalendar[selectedDay].numberOfTasks = tasksToAdd;

        for (int taskNumber = 0; taskNumber < tasksToAdd; taskNumber++) {
            printf("Enter task %d: ", taskNumber + 1);

            fgets(
                weekCalendar[selectedDay].taskList[taskNumber],
                MAX_TASK_LENGTH,
                stdin
            );

            weekCalendar[selectedDay]
                .taskList[taskNumber][strcspn(
                    weekCalendar[selectedDay].taskList[taskNumber],
                    "\n")] = '\0';
        }
    }

    printf("\n\n===== WEEKLY TASK SCHEDULE =====\n");

    for (int dayIndex = 0; dayIndex < TOTAL_DAYS; dayIndex++) {
        printf("\n%s:\n", weekCalendar[dayIndex].dayName);

        if (weekCalendar[dayIndex].numberOfTasks == 0) {
            printf("  No tasks scheduled.\n");
        } else {
            for (int taskNumber = 0;
                 taskNumber < weekCalendar[dayIndex].numberOfTasks;
                 taskNumber++) {

                printf("  %d. %s\n",
                       taskNumber + 1,
                       weekCalendar[dayIndex].taskList[taskNumber]);
            }
        }
    }

    return 0;
}

2. even & odd

#include <stdio.h>

void arrangeEvenOdd(int *arr, int size) {
    int *current, *position, *shift;
    int evenValue;

    position = arr;   // Position where next even number should be placed

    for (current = arr; current < arr + size; current++) {
        if (*current % 2 == 0) {
            evenValue = *current;

            // Shift elements right to preserve order
            for (shift = current; shift > position; shift--) {
                *shift = *(shift - 1);
            }

            *position = evenValue;
            position++;
        }
    }
}

int main() {
    int numbers[] = {3, 8, 5, 2, 9, 6, 1, 4};
    int size = sizeof(numbers) / sizeof(numbers[0]);

    arrangeEvenOdd(numbers, size);

    printf("Rearranged Array: ");
    int *ptr;
    for (ptr = numbers; ptr < numbers + size; ptr++) {
        printf("%d ", *ptr);
    }

    return 0;
}


Output:
Rearranged Array: 8 2 6 4 3 5 9 1


3. key value finder

#include <stdio.h>

int searchKey(int matrix[][100], int n, int key) {
    int row = 0;
    int col = n - 1;

    while (row < n && col >= 0) {
        if (matrix[row][col] == key) {
            return 1; // Key found
        }
        else if (matrix[row][col] > key) {
            col--; // Move left
        }
        else {
            row++; // Move down
        }
    }

    return 0; // Key not found
}

int main() {
    int n = 4;
    int key = 29;

    int matrix[100][100] = {
        {10, 20, 30, 40},
        {15, 25, 35, 45},
        {27, 29, 37, 48},
        {32, 33, 39, 50}
    };

    if (searchKey(matrix, n, key))
        printf("Key Found");
    else
        printf("Key Not Found");

    return 0;
}


Output: Key found
